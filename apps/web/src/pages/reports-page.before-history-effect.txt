function formatNumber(
  value:
    | number
    | null
    | undefined,
) {
  return Number(
    value ?? 0,
  ).toLocaleString();
}

function formatPercent(
  value:
    | number
    | null
    | undefined,
) {
  return `${Number(
    value ?? 0,
  ).toFixed(1)}%`;
}

function formatDateTime(
  value:
    | string
    | null
    | undefined,
) {
  if (!value) {
    return '-';
  }

  const date =
    new Date(
      value,
    );

  if (
    Number.isNaN(
      date.getTime(),
    )
  ) {
    return '-';
  }

  return date.toLocaleString();
}

function formatReportDate(
  value:
    | string
    | null
    | undefined,
) {
  if (!value) {
    return '-';
  }

  const date =
    new Date(
      value,
    );

  if (
    Number.isNaN(
      date.getTime(),
    )
  ) {
    return value;
  }

  return date.toLocaleDateString(
    undefined,
    {
      year:
        'numeric',

      month:
        'short',

      day:
        'numeric',
    },
  );
}

function toDateInputValue(
  date: Date,
) {
  const year =
    date.getFullYear();

  const month =
    String(
      date.getMonth() +
      1,
    ).padStart(
      2,
      '0',
    );

  const day =
    String(
      date.getDate(),
    ).padStart(
      2,
      '0',
    );

  return `${year}-${month}-${day}`;
}

function startOfWeek(
  date: Date,
) {
  const result =
    new Date(
      date,
    );

  const currentDay =
    result.getDay();

  const difference =
    currentDay === 0
      ? -6
      : 1 - currentDay;

  result.setDate(
    result.getDate() +
    difference,
  );

  return result;
}

function downloadBlob(
  blob: Blob,
  filename: string,
) {
  const url =
    window.URL.createObjectURL(
      blob,
    );

  const anchor =
    document.createElement(
      'a',
    );

  anchor.href =
    url;

  anchor.download =
    filename;

  document.body.appendChild(
    anchor,
  );

  anchor.click();

  anchor.remove();

  window.URL.revokeObjectURL(
    url,
  );
}

export function ReportsPage() {
  // ==========================================================================
  // AUTH
  // ==========================================================================

  const user =
    useAuthStore(
      (state) =>
        state.user,
    );

  // ==========================================================================
  // SHARED FILTER STATE
  // ==========================================================================

  const regionId =
    useFilterStore(
      (state) =>
        state.regionId,
    );

  const districtId =
    useFilterStore(
      (state) =>
        state.districtId,
    );

  const branchId =
    useFilterStore(
      (state) =>
        state.branchId,
    );

  const productId =
    useFilterStore(
      (state) =>
        state.productId,
    );

  const startDate =
    useFilterStore(
      (state) =>
        state.startDate,
    );

  const endDate =
    useFilterStore(
      (state) =>
        state.endDate,
    );

  const setStartDate =
    useFilterStore(
      (state) =>
        state.setStartDate,
    );

  const setEndDate =
    useFilterStore(
      (state) =>
        state.setEndDate,
    );
  const applyReportFilters =
    useFilterStore(
      (state) =>
        state.applyReportFilters,
    );
  // ==========================================================================
  // LOCAL STATE
  // ==========================================================================
  // Keep the staff selection local to Reports.
  // Store the branch alongside the staff ID to prevent
  // a stale selection when the organization filter changes.
  const [
    staffSelection,
    setStaffSelection,
  ] = useState<{
    branchId: string;
    staffId: string;
  } | null>(null);
  const queryClient = useQueryClient();

  const refreshReportHistory = () => {
    void queryClient.invalidateQueries({
      queryKey: ['reports', 'history'],
    });
  };
  const selectedStaffId =
    staffSelection?.branchId === branchId
      ? staffSelection.staffId
      : '';
  const [
    periodPreset,
    setPeriodPreset,
  ] =
    useState<ReportPeriodPreset>(
      'CUSTOM',
    );

  const [
    exportingCsv,
    setExportingCsv,
  ] =
    useState(
      false,
    );

  const [
    exportingXlsx,
    setExportingXlsx,
  ] =
    useState(
      false,
    );

  const [
    exportError,
    setExportError,
  ] =
    useState<
      string | null
    >(null);
  const [
    savedReportError,
    setSavedReportError,
  ] = useState<string | null>(null);

  const [
    runningSavedReportId,
    setRunningSavedReportId,
  ] = useState<string | null>(null);
  // ==========================================================================
  // PERMISSIONS
  // ==========================================================================

  /*
   * Frontend authorization is only for UX.
   * Backend report.export permission remains authoritative.
   */
  const canExport =
    user?.roles.some(
      (role) =>
        role ===
        'SUPER_ADMIN' ||
        role ===
        'SYSTEM_ADMIN' ||
        role ===
        'REGIONAL_MANAGER' ||
        role ===
        'REGIONAL_ANALYST' ||
        role ===
        'DISTRICT_MANAGER' ||
        role ===
        'DISTRICT_ANALYST' ||
        role ===
        'BRANCH_MANAGER' ||
        role ===
        'AUDITOR',
    ) ?? false;
  // ==========================================================================
  // STAFF OPTIONS
  // ==========================================================================

  const staffOptionsResult = useQuery({
    queryKey: [
      'report-staff-options',
      user?.id,
      branchId,
    ],

    queryFn: () =>
      getStaffOptionsApi(branchId!),

    enabled:
      Boolean(user && branchId),
  });

  const staffOptions =
    staffOptionsResult.data ?? [];

  const selectedStaffIsAvailable =
    selectedStaffId !== '' &&
    staffOptionsResult.isSuccess &&
    staffOptions.some(
      (staff) =>
        staff.staffId === selectedStaffId,
    );

  // Only include a selected staff ID after the
  // options for the current branch have been verified.
  const effectiveStaffId =
    selectedStaffIsAvailable
      ? selectedStaffId
      : '';
  // ==========================================================================
  // REPORT QUERY
  // ==========================================================================

  const reportQuery =
    useMemo<ReportQuery>(
      () => ({
        ...(regionId
          ? {
            regionId,
          }
          : {}),

        ...(districtId
          ? {
            districtId,
          }
          : {}),

        ...(branchId
          ? {
            branchId,
          }
          : {}),
        ...(effectiveStaffId
          ? {
            staffId: effectiveStaffId,
          }
          : {}),
        ...(productId
          ? {
            productId,
          }
          : {}),

        ...(startDate
          ? {
            startDate,
          }
          : {}),

        ...(endDate
          ? {
            endDate,
          }
          : {}),
      }),
      [
        regionId,
        districtId,
        branchId,
        effectiveStaffId,
        productId,
        startDate,
        endDate,
      ],
    );
  const staffSelectionPending =
    selectedStaffId !== '' &&
    !selectedStaffIsAvailable;

  const staffSelectionInvalid =
    selectedStaffId !== '' &&
    staffOptionsResult.isSuccess &&
    !selectedStaffIsAvailable;

  const canRunReport =
    Boolean(user) &&
    !staffSelectionPending;
const reportResult = useQuery({
  queryKey: [
    'performance-report',
    user?.id,
    reportQuery,
  ],

  queryFn: async () => {
    const result =
      await getPerformanceReportApi(
        reportQuery,
      );

    refreshReportHistory();

    return result;
  },

  enabled: canRunReport,

  // Avoid unnecessary report generation
  // when the browser reconnects or remounts.
  refetchOnWindowFocus: false,
  refetchOnReconnect: false,
  refetchOnMount: false,

  // Avoid repeated requests that could
  // create unnecessary audit records.
  retry: false,

  // Reuse an existing report result
  // for five minutes.
  staleTime: 5 * 60 * 1000,
});

const report = reportResult.data;
  // ==========================================================================
  // SAVED REPORTS
  // ==========================================================================

  const handleLoadSavedReport = (
    savedReport: SavedReport,
  ) => {
    setSavedReportError(null);

    const filters = savedReport.filters;

    applyReportFilters({
      regionId: filters.regionId,
      districtId: filters.districtId,
      branchId: filters.branchId,
      productId: filters.productId,
      startDate: filters.startDate,
      endDate: filters.endDate,
    });

    setStaffSelection(
      filters.staffId && filters.branchId
        ? {
          branchId: filters.branchId,
          staffId: filters.staffId,
        }
        : null,
    );

    setPeriodPreset('CUSTOM');
    setExportError(null);
  };

  const handleRunSavedReport = async (
    savedReport: SavedReport,
  ) => {
    if (runningSavedReportId) {
      return;
    }

    setSavedReportError(null);
    setRunningSavedReportId(savedReport.id);

    try {
      // The backend revalidates the user's scope
      // before executing the saved report.
      await runSavedReportApi(savedReport.id);

      refreshReportHistory();

      // Restore filters so the existing Reports page
      // displays the selected saved report.
      handleLoadSavedReport(savedReport);
    } catch {
      setSavedReportError(
        'Unable to run the saved report. Check your permissions and saved filters.',
      );
    } finally {
      setRunningSavedReportId(null);
    }
  };
  // ==========================================================================
  // PERIOD PRESETS
  // ==========================================================================

  const handlePeriodPreset = (
    _event: MouseEvent<HTMLElement>,
    value:
      ReportPeriodPreset | null,
  ) => {
    if (!value) {
      return;
    }

    setPeriodPreset(
      value,
    );

    if (
      value ===
      'CUSTOM'
    ) {
      return;
    }

    const now =
      new Date();

    let start =
      new Date(
        now,
      );

    const end =
      new Date(
        now,
      );

    switch (value) {
      case 'DAILY':
        start =
          new Date(
            now,
          );

        break;

      case 'WEEK':
        start =
          startOfWeek(
            now,
          );

        break;

      case 'MONTH':
        start =
          new Date(
            now.getFullYear(),
            now.getMonth(),
            1,
          );

        break;

      case 'YTD':
        start =
          new Date(
            now.getFullYear(),
            0,
            1,
          );

        break;

      default:
        return;
    }

    setStartDate(
      toDateInputValue(
        start,
      ),
    );

    setEndDate(
      toDateInputValue(
        end,
      ),
    );
  };

  // ==========================================================================
  // EXPORT CSV
  // ==========================================================================

  const handleCsvExport =
    async () => {
      setExportError(
        null,
      );

      setExportingCsv(
        true,
      );

      try {
        const blob =
          await exportPerformanceCsvApi(
            reportQuery,
          );

        refreshReportHistory();

        downloadBlob(
          blob,
          `performance-report-${startDate || 'start'}-${endDate || 'end'}.csv`,
        );
      } catch {
        setExportError(
          'Unable to export the CSV report.',
        );
      } finally {
        setExportingCsv(
          false,
        );
      }
    };
  if (!canRunReport) {
    return;
  }
  // ==========================================================================
  // EXPORT XLSX
  // ==========================================================================

  const handleXlsxExport =
    async () => {
      setExportError(
        null,
      );

      setExportingXlsx(
        true,
      );

      try {
        const blob =
          await exportPerformanceXlsxApi(
            reportQuery,
          );

        refreshReportHistory();

        downloadBlob(
          blob,
          `performance-report-${startDate || 'start'}-${endDate || 'end'}.xlsx`,
        );
      } catch {
        setExportError(
          'Unable to export the Excel report.',
        );
      } finally {
        setExportingXlsx(
          false,
        );
      }
    };
  if (!canRunReport) {
    return;
  }
  // ==========================================================================
  // LOADING
  // ==========================================================================

  if (
    reportResult.isLoading
  ) {
    return (
      <Box
        sx={{
          minHeight:
            400,

          display:
            'flex',

          alignItems:
            'center',

          justifyContent:
            'center',
        }}
      >
        <CircularProgress />
      </Box>
    );
  }

  // ==========================================================================
  // LOAD ERROR
  // ==========================================================================

  if (reportResult.isError) {
    return (
      <Stack spacing={3}>
        <Typography
          variant="h4"
          sx={{ fontWeight: 700 }}
        >
          Performance Reports
        </Typography>

        {user && (
          <SavedReportsPanel
            userId={user.id}
            currentFilters={reportQuery}
            canSave={false}
            onLoad={handleLoadSavedReport}
            onRun={(savedReport) => {
              void handleRunSavedReport(savedReport);
            }}
          />
        )}

        {runningSavedReportId && (
          <Alert severity="info">
            Running saved report...
          </Alert>
        )}

        {savedReportError && (
          <Alert severity="error">
            {savedReportError}
          </Alert>
        )}

        <Alert severity="error">
          Unable to load the performance report.
          You can load another saved report
          or retry the current request.
        </Alert>

        <Button
          variant="outlined"
          startIcon={<RefreshOutlined />}
          onClick={() => {
            void reportResult.refetch();
          }}
          sx={{ alignSelf: 'flex-start' }}
          disabled={!canRunReport || reportResult.isFetching}
        >
          Retry
        </Button>
      </Stack>
    );
  }

  if (!report) {
    return null;
  }

  const summary =
    report.summary;

  // ==========================================================================
  // TARGET DISPLAY POLICY
  //
  // Match the Dashboard's existing behavior.
  // A positive target means an applicable target exists.
  // Zero currently represents "Not assigned" in the API.
  // Monthly targets are not prorated.
  // ==========================================================================

  const hasAssignedTarget =
    Number(summary.achievement.target) > 0;

  const targetDisplayValue = hasAssignedTarget
    ? formatNumber(summary.achievement.target)
    : 'Not assigned';

  const achievementDisplayValue = hasAssignedTarget
    ? formatPercent(summary.achievement.percentage)
    : 'N/A';
  // ==========================================================================
  // RENDER
  // ==========================================================================

  return (
    <Stack
      spacing={3}
    >
      {/* =====================================================================
          HEADER
      ===================================================================== */}

      <Stack
        direction={{
          xs:
            'column',

          sm:
            'row',
        }}
        spacing={2}
        sx={{
          justifyContent:
            'space-between',

          alignItems: {
            xs:
              'stretch',

            sm:
              'center',
          },
        }}
      >
        <Box>
          <Typography
            variant="h4"
            sx={{
              fontWeight:
                700,
            }}
          >
            Performance Reports
          </Typography>

          <Typography
            variant="body2"
            color="text.secondary"
            sx={{
              mt:
                0.5,
            }}
          >
            Generate, review and export digital performance reports
            across your authorized organizational scope.
          </Typography>
        </Box>

        <Stack
          direction="row"
          spacing={1}
          sx={{
            flexWrap:
              'wrap',

            rowGap:
              1,

            justifyContent: {
              xs:
                'flex-start',

              sm:
                'flex-end',
            },
          }}
        >
          <Button
            variant="outlined"
            startIcon={
              <RefreshOutlined />
            }
            onClick={() =>
              reportResult.refetch()
            }
            disabled={
              reportResult.isFetching ||
              !canRunReport
            }
          >
            {reportResult.isFetching
              ? 'Refreshing...'
              : 'Refresh'}
          </Button>

          {canExport && (
            <>
              <Button
                variant="outlined"
                startIcon={
                  <DownloadOutlined />
                }
                onClick={
                  handleCsvExport
                }
                disabled={
                  exportingCsv ||
                  exportingXlsx ||
                  !canRunReport
                }
              >
                {exportingCsv
                  ? 'Exporting...'
                  : 'CSV'}
              </Button>

              <Button
                variant="contained"
                startIcon={
                  <TableChartOutlined />
                }
                onClick={
                  handleXlsxExport
                }
                disabled={
                  exportingCsv ||
                  exportingXlsx
                }
              >
                {exportingXlsx
                  ? 'Exporting...'
                  : 'Excel'}
              </Button>
            </>
          )}
        </Stack>
      </Stack>
      {/* =====================================================================
    SAVED REPORTS
===================================================================== */}
      <ReportHistoryPanel />
      {user && (

        <SavedReportsPanel
          userId={user.id}
          currentFilters={reportQuery}
          canSave={
            canRunReport &&
            !reportResult.isFetching &&
            !runningSavedReportId
          }
          onLoad={handleLoadSavedReport}
          onRun={(savedReport) => {
            void handleRunSavedReport(savedReport);
          }}
        />
      )}

      {runningSavedReportId && (
        <Alert severity="info">
          Running saved report...
        </Alert>

      {savedReportError && (
        <Alert severity="error">
          {savedReportError}
        </Alert>
      )}
      {/* =====================================================================
          REPORT PERIOD PRESETS
      ===================================================================== */}

      <Paper
        variant="outlined"
        sx={{
          p:
            2,
        }}
      >
        <Stack
          direction={{
            xs:
              'column',

            md:
              'row',
          }}
          spacing={2}
          sx={{
            alignItems: {
              xs:
                'stretch',

              variant="subtitle1"

                fontWeight:
                  600,
              }}

              Report Period

              variant="body2"
              color="text.secondary"

              Select a predefined reporting period or use custom dates.
            </Typography>

            size="small"

              periodPreset

              handlePeriodPreset
            }

              flexWrap:
                'wrap',

              rowGap:
                1,
            }}
          >
            <ToggleButton
              value="DAILY"
            >
              Daily
            </ToggleButton>

            <ToggleButton
              value="WEEK"
            >
              This Week
            </ToggleButton>

            <ToggleButton
              value="MONTH"
            >
              This Month
            </ToggleButton>

            <ToggleButton
              value="YTD"
            >
              YTD
            </ToggleButton>

            <ToggleButton
              value="CUSTOM"
            >
              Custom
            </ToggleButton>
          </ToggleButtonGroup>
        </Stack>
      </Paper>

      {/* =====================================================================
          ORGANIZATION / PRODUCT / CUSTOM DATE FILTERS
      ===================================================================== */}

      <Paper
        variant="outlined"
        sx={{
          p:
            2,
        }}
      >
        <Stack spacing={2}>
          <OrganizationFilter />

          <Box
            sx={{
              maxWidth: 360,
            }}
          >
            <FormControl
              fullWidth
              size="small"
              disabled={
                !branchId ||
                staffOptionsResult.isPending ||
                staffOptionsResult.isFetching ||
                staffOptionsResult.isError
              }
            >
              <InputLabel id="report-staff-label">
                Staff
              </InputLabel>

              <Select
                labelId="report-staff-label"
                label="Staff"
                value={selectedStaffId}
                onChange={(event) => {
                  const nextStaffId =
                    event.target.value;

                  setStaffSelection(
                    nextStaffId && branchId
                      ? {
                        branchId,
                        staffId: nextStaffId,
                      }
                      : null,
                  );
                }}
              >
                <MenuItem value="">
                  All Staff
                </MenuItem>

                {staffOptions.map((staff) => {
                  const fullName = [
                    staff.firstName,
                    staff.middleName,
                    staff.lastName,
                  ]
                    .filter(Boolean)
                    .join(' ');

                  return (
                    <MenuItem
                      key={staff.staffId}
                      value={staff.staffId}
                    >
                      {fullName}
                      {staff.employeeNumber
                        ? ` (${staff.employeeNumber})`
                        : ''}
                    </MenuItem>
                  );
                })}
              </Select>
            </FormControl>

            {!branchId && (
              <Typography
                variant="caption"
                color="text.secondary"
              >
                Select a branch to filter by staff.
              </Typography>
            )}

            {branchId &&
              staffOptionsResult.isFetching && (
                <Typography
                  variant="caption"
                  color="text.secondary"
                >
                  Loading staff options...
                </Typography>
              )}

            {branchId &&
              staffOptionsResult.isError && (
                <Alert
                  severity="error"
                  sx={{ mt: 1 }}
                >
                  Unable to load staff options.
                </Alert>
              )}

            {/* Invalid staff selection warning */}
            {staffSelectionInvalid && (
              <Alert
                severity="warning"
                sx={{ mt: 1 }}
              >
                The selected staff member is no longer
                available in this branch. Select All Staff
                or choose another staff member.
              </Alert>
            )}
          </Box>
        </Stack>
      </Paper>

      {/* =====================================================================
          EXPORT ERROR
      ===================================================================== */}

      {exportError && (
        <Alert
          severity="error"
          onClose={() =>
            setExportError(
              null,
            )
          }
        >
          {exportError}
        </Alert>
      )}

      {/* =====================================================================
          REPORT INFORMATION
      ===================================================================== */}
